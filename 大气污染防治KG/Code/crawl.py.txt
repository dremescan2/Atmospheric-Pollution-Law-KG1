import requests
from bs4 import BeautifulSoup
import os
import time

# 保存路径
save_path = "../Data/Raw_Data/"
os.makedirs(save_path, exist_ok=True)

# 大气污染防治核心法规 & 标准 URL
urls = [
    "https://www.mee.gov.cn/ywgz/fgbz/fl/201807/t20180702_449582.html",
    "https://www.mee.gov.cn/hjbhbz/bzwb/dqhjbh/dqhjzjbz/201203/t20120301_226489.html"
]

headers = {
    "User-Agent": "Mozilla/5.0"
}

def crawl():
    for url in urls:
        try:
            res = requests.get(url, headers=headers, timeout=20)
            res.encoding = "utf-8"
            soup = BeautifulSoup(res.text, "html.parser")

            title = soup.find("h1").text.strip()
            content = soup.find("div", class_="content").text.strip()

            with open(f"{save_path}{title}.txt", "w", encoding="utf-8") as f:
                f.write(content)

            print(f"已保存：{title}")
            time.sleep(1)
        except Exception as e:
            print("爬取失败：", e)

if __name__ == "__main__":
    crawl()
