import json
import networkx as nx
import matplotlib.pyplot as plt

plt.rcParams["font.sans-serif"] = ["SimHei"]

with open("../Data/KG_Data/entities.json", encoding="utf-8") as f:
    entities = json.load(f)
with open("../Data/KG_Data/relations.json", encoding="utf-8") as f:
    relations = json.load(f)

G = nx.Graph()

for e in entities:
    G.add_node(e["name"], type=e["type"])
for r in relations:
    G.add_edge(r["source"], r["target"], label=r["relation"])

pos = nx.spring_layout(G, seed=123)
nx.draw(G, pos, with_labels=True, node_size=1000, font_size=8)
plt.savefig("../Data/KG_Data/kg.png", dpi=300)
plt.show()
