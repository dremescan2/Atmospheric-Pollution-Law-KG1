import re
import os

raw_dir = "../Data/Raw_Data/"
out_dir = "../Data/Processed_Data/"
os.makedirs(out_dir, exist_ok=True)

def clean_all():
    for fname in os.listdir(raw_dir):
        path = os.path.join(raw_dir, fname)
        with open(path, "r", encoding="utf-8") as f:
            text = f.read()

        # 清洗
        text = re.sub(r"\s+", " ", text)
        text = re.sub(r"【.*?】|〔.*?〕", "", text)

        # 按条款切分
        clauses = re.split(r"第[一二三四五六七八九十\d]+条|第一章|第二章", text)
        clauses = [c.strip() for c in clauses if len(c.strip()) > 10]

        out_path = os.path.join(out_dir, fname.replace(".txt", "_clean.txt"))
        with open(out_path, "w", encoding="utf-8") as f:
            for i, c in enumerate(clauses):
                f.write(f"条款{i+1}：{c}\n")

    print("清洗完成！")

if __name__ == "__main__":
    clean_all()
